# NAVIA PATH - INSTALL / UPDATE ROBOT
# Usage:
# 1) Ouvrir ce TXT depuis le RPA_SHARE.
# 2) Copier tout le contenu.
# 3) Coller directement dans PowerShell du compte robot.
#
# Aucun .ps1 n'est execute depuis le partage reseau.
# Compatible avec le principe RemoteSigned / ConstrainedLanguage :
# uniquement des cmdlets/commandes PowerShell standard + robocopy.

$ErrorActionPreference = "Stop"

$ShareRoot = "\\svm-prod1.rsi.prd.mlb.nbyt.fr\RPA_SHARE\Imad\NavIA Pass"
$LocalRoot = "$env:LOCALAPPDATA\NAVIA_PATH"
$Provider = "vertex"

$ShareVersionFile = "$ShareRoot\CURRENT_VERSION.txt"

Write-Host ""
Write-Host "============================================================" -ForegroundColor Cyan
Write-Host " NAVIA PATH - ROBOT INSTALL / UPDATE"
Write-Host "============================================================" -ForegroundColor Cyan
Write-Host "Robot    : $env:USERNAME"
Write-Host "Share    : $ShareRoot"
Write-Host "Local    : $LocalRoot"
Write-Host "Provider : $Provider"
Write-Host ""

# 1 - Verifier le share et le manifeste de release
if (-not (Test-Path -LiteralPath $ShareRoot)) {
    throw "RPA_SHARE inaccessible : $ShareRoot"
}

if (-not (Test-Path -LiteralPath $ShareVersionFile)) {
    throw "CURRENT_VERSION.txt absent : $ShareVersionFile"
}

$Meta = Get-Content -LiteralPath $ShareVersionFile

$Version = ($Meta | Where-Object { $_ -like "VERSION=*" }) -replace "^VERSION=", ""
$ExeName = ($Meta | Where-Object { $_ -like "EXE=*" }) -replace "^EXE=", ""
$ExpectedHash = ($Meta | Where-Object { $_ -like "SHA256=*" }) -replace "^SHA256=", ""

if (-not $Version) { throw "VERSION absente de CURRENT_VERSION.txt" }
if (-not $ExeName) { throw "EXE absent de CURRENT_VERSION.txt" }
if (-not $ExpectedHash) { throw "SHA256 absent de CURRENT_VERSION.txt" }

$ShareExe = "$ShareRoot\$ExeName"

if (-not (Test-Path -LiteralPath $ShareExe)) {
    throw "Executable publie introuvable : $ShareExe"
}

Write-Host "[SHARE] Version : $Version" -ForegroundColor Green
Write-Host "[SHARE] EXE     : $ExeName"
Write-Host "[SHARE] SHA256  : $ExpectedHash"
Write-Host ""

# 2 - Creer le runtime local
if (-not (Test-Path -LiteralPath $LocalRoot)) {
    New-Item -ItemType Directory -Path $LocalRoot -Force | Out-Null
}

foreach ($FolderName in @("Runs","Logs","Cache","EdgeProfile","ChromeProfile")) {
    $Folder = "$LocalRoot\$FolderName"
    if (-not (Test-Path -LiteralPath $Folder)) {
        New-Item -ItemType Directory -Path $Folder -Force | Out-Null
    }
}

# 3 - Verifier / synchroniser l'EXE versionne
$LocalExe = "$LocalRoot\$ExeName"
$NeedExeCopy = $true

if (Test-Path -LiteralPath $LocalExe) {
    $LocalHash = (Get-FileHash -LiteralPath $LocalExe -Algorithm SHA256).Hash

    if ($LocalHash -eq $ExpectedHash) {
        $NeedExeCopy = $false
        Write-Host "[EXE] Version locale deja correcte : $ExeName" -ForegroundColor Green
        Write-Host "[EXE] SHA256 : OK" -ForegroundColor Green
    }
    else {
        Write-Host "[EXE] Hash local different -> nouvelle copie requise." -ForegroundColor Yellow
    }
}
else {
    Write-Host "[EXE] Release locale absente -> installation requise." -ForegroundColor Yellow
}

if ($NeedExeCopy) {
    $TempExe = "$LocalRoot\$ExeName.download"

    Remove-Item -LiteralPath $TempExe -Force -ErrorAction SilentlyContinue

    Write-Host "[EXE] Copie locale depuis le RPA_SHARE..." -ForegroundColor Cyan
    Copy-Item -LiteralPath $ShareExe -Destination $TempExe -Force

    $CopiedHash = (Get-FileHash -LiteralPath $TempExe -Algorithm SHA256).Hash

    if ($CopiedHash -ne $ExpectedHash) {
        Remove-Item -LiteralPath $TempExe -Force -ErrorAction SilentlyContinue
        throw "SHA256 incorrect apres copie. Installation annulee."
    }

    Move-Item -LiteralPath $TempExe -Destination $LocalExe -Force
    Write-Host "[EXE] Copie terminee et SHA256 valide." -ForegroundColor Green
}

Unblock-File -LiteralPath $LocalExe -ErrorAction SilentlyContinue

# 4 - Toujours resynchroniser la configuration.
#     Cela permet de mettre a jour .env.vertex sans changer la version du binaire.
$ShareConfig = "$ShareRoot\config"
$LocalConfig = "$LocalRoot\config"

if (-not (Test-Path -LiteralPath $ShareConfig)) {
    throw "Dossier config absent du RPA_SHARE : $ShareConfig"
}

Write-Host "[CONFIG] Synchronisation..." -ForegroundColor Cyan

robocopy `
    $ShareConfig `
    $LocalConfig `
    /E `
    /COPY:DAT `
    /R:2 `
    /W:2 | Out-Null

$ConfigRoboCode = $LASTEXITCODE

if ($ConfigRoboCode -ge 8) {
    throw "Echec de synchronisation config. Robocopy code : $ConfigRoboCode"
}

Write-Host "[CONFIG] OK" -ForegroundColor Green

# 5 - Copier le manifeste courant
Copy-Item `
    -LiteralPath $ShareVersionFile `
    -Destination "$LocalRoot\CURRENT_VERSION.txt" `
    -Force

# 6 - Compatibilite UiPath.
#     La release reste versionnee, mais les XAML existants savent deja chercher :
#     %LOCALAPPDATA%\NAVIA_PATH\NAVIA_PATH.exe
$UiPathExe = "$LocalRoot\NAVIA_PATH.exe"

$NeedStableExe = $true

if (Test-Path -LiteralPath $UiPathExe) {
    $StableHash = (Get-FileHash -LiteralPath $UiPathExe -Algorithm SHA256).Hash

    if ($StableHash -eq $ExpectedHash) {
        $NeedStableExe = $false
    }
}

if ($NeedStableExe) {
    Copy-Item -LiteralPath $LocalExe -Destination $UiPathExe -Force
}

Unblock-File -LiteralPath $UiPathExe -ErrorAction SilentlyContinue

$StableHashFinal = (Get-FileHash -LiteralPath $UiPathExe -Algorithm SHA256).Hash

if ($StableHashFinal -ne $ExpectedHash) {
    throw "NAVIA_PATH.exe local ne correspond pas a la release $Version."
}

Write-Host "[UIPATH] NAVIA_PATH.exe correspond a v$Version : OK" -ForegroundColor Green

# 7 - Nettoyer les anciens EXE versionnes seulement apres validation du nouveau
Get-ChildItem `
    -LiteralPath $LocalRoot `
    -Filter "NAVIA_PATH_v*.exe" `
    -File `
    -ErrorAction SilentlyContinue |
    Where-Object { $_.Name -ne $ExeName } |
    Remove-Item -Force -ErrorAction SilentlyContinue

# 8 - Afficher la version reellement installee
Write-Host ""
Write-Host "============================================================" -ForegroundColor Cyan
Write-Host " VERSION LOCALE"
Write-Host "============================================================" -ForegroundColor Cyan

& $UiPathExe --version

# 9 - Check reel depuis l'EXE LOCAL uniquement
Write-Host ""
Write-Host "============================================================" -ForegroundColor Cyan
Write-Host " CHECK $Provider"
Write-Host "============================================================" -ForegroundColor Cyan

Push-Location -LiteralPath $LocalRoot

try {
    & $UiPathExe check --provider $Provider
    $CheckCode = $LASTEXITCODE
}
finally {
    Pop-Location
}

if ($CheckCode -ne 0) {
    throw "NAVIA PATH check a echoue. Code : $CheckCode"
}

Write-Host ""
Write-Host "============================================================" -ForegroundColor Green
Write-Host " NAVIA PATH READY"
Write-Host "============================================================" -ForegroundColor Green
Write-Host "Robot        : $env:USERNAME"
Write-Host "Version      : $Version"
Write-Host "Release EXE  : $LocalExe"
Write-Host "UiPath EXE   : $UiPathExe"
Write-Host "Config locale: $LocalConfig"
Write-Host ""
Write-Host "Le robot est synchronise avec la release courante du RPA_SHARE." -ForegroundColor Green
